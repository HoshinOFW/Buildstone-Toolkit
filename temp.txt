import groovy.io.FileType

// helpers
def isVersionCommon = { Project p ->
    // matches :1.20.1:common, :1.21.1:common, :1.21.11:common, etc.
    def parts = p.path.split(':').findAll { it }
    parts.size() == 2 && parts[1] == 'common' && (parts[0] ==~ /\d+(\.\d+){1,3}/)
}

def parseVer = { String v -> v.split(/\./).collect { it as int } }
def compareVer = { String a, String b ->
    def pa = parseVer(a); def pb = parseVer(b)
    def n = Math.max(pa.size(), pb.size())
    for (int i = 0; i < n; i++) {
        def ai = i < pa.size() ? pa[i] : 0
        def bi = i < pb.size() ? pb[i] : 0
        if (ai != bi) return ai <=> bi
    }
    0
}

def relFileSet = { File root ->
    def out = new LinkedHashSet<String>()
    if (!root.exists()) return out
    root.eachFileRecurse(FileType.FILES) { f ->
        def rel = root.toPath().relativize(f.toPath()).toString().replace('\\','/')
        out.add(rel)
    }
    out
}

def javaRelSet = { Project prj -> relFileSet(prj.file("src/main/java")) }

import groovy.json.JsonSlurper
import groovy.io.FileType

// --- broken class excludes (version -> patterns relative to src/main/java) ---
def loadBrokenMap = {
    File f = rootProject.file("excludeClassList.json")
    if (!f.exists()) return [:]

    def parsed = new JsonSlurper().parse(f)
    if (!(parsed instanceof Map)) {
        throw new GradleException("excludeClassList.json must contain a JSON object: { \"1.21.1\": [\"...\"], ... }")
    }

    // normalize to Map<String, List<String>>
    def out = [:].withDefault { [] }
    parsed.each { k, v ->
        if (v == null) return
        def ver = k.toString()
        def list = (v instanceof Collection) ? v : [v]
        out[ver] = list.collect { it.toString().replace('\\','/').trim() }.findAll { it }
    }
    return out
}

// patterns accepted by Gradle FileTree.exclude are Ant-style.
// We'll treat provided entries as "relative to src/main/java".
def normalizeExcludePatterns = { Collection<String> rels ->
    rels.collect { p ->
        def s = p.replace('\\','/').trim()
        // Ensure it's relative-ish; you can keep it simple:
        // - "com/x/Y.java" works directly
        // - "**/com/x/Y.java" also works
        // If user gives "Y.java", we can make it match anywhere:
        if (!s.contains("/")) {
            return "**/${s}"
        }
        return s
    }
}


// ---- patch chain wiring ----
gradle.projectsEvaluated {
    def sharedJava = rootProject.file("common/src/main/java")
    def sharedRes  = rootProject.file("common/src/main/resources")

    def commons = rootProject.subprojects.findAll { isVersionCommon(it) }

    def verToProj = [:]
    commons.each { p ->
        def ver = p.path.split(':').findAll { it }[0]
        verToProj[ver] = p
    }

    def versions = verToProj.keySet().toList().sort { a, b -> compareVer(a, b) }

    if (versions.size() <= 1) {
        logger.lifecycle("[patch-common] Only one version common found; no patching needed.")
        return
    }

    def brokenByVerRaw = loadBrokenMap()
    def brokenByVer = [:].withDefault { [] as List<String> }
    brokenByVerRaw.each { ver, list ->
        brokenByVer[ver] = normalizeExcludePatterns(list)
    }

// cumulative broken excludes so newer versions inherit older broken declarations
    def cumulativeBroken = [:].withDefault { new LinkedHashSet<String>() }
    def running = new LinkedHashSet<String>()
    versions.each { ver ->
        running.addAll(brokenByVer[ver] ?: [])
        cumulativeBroken[ver] = new LinkedHashSet<String>(running)
    }


    // Chain:
    for (int i = 1; i < versions.size(); i++) {
        def patchVer = versions[i]
        def brokenExcludesForPatch = cumulativeBroken[patchVer]
        def baseVer  = versions[i - 1]

        def patchP = verToProj[patchVer]
        def baseP  = verToProj[baseVer]

        patchP.logger.lifecycle("[patch-common] ${patchP.path} patches ${baseP.path}")

        def patchJava = patchP.file("src/main/java")
        def patchRes  = patchP.file("src/main/resources")

        // ---------- Java: compile while all the way back (shared -> all previous -> patch)h ----------
        patchP.tasks.withType(JavaCompile).configureEach { JavaCompile jc ->
            if (jc.name != "compileJava") return

            def trees = []
            trees << patchP.fileTree(sharedJava) { include("**/*.java") }

            // Precompute relpaths for each version project once
            def relByVer = [:]
            versions.each { v -> relByVer[v] = javaRelSet(verToProj[v]) }

            // older layers (exclude overrides + broken)
            for (int j = 0; j < i; j++) {
                def v = versions[j]
                def proj = verToProj[v]
                def layerJava = proj.file("src/main/java")

                def overriddenByLater = new LinkedHashSet<String>()
                for (int k = j + 1; k <= i; k++) {
                    def key = versions[k]
                    def s = relByVer[key]
                    if (s == null) {
                        patchP.logger.lifecycle("[patch-common] WARNING: no rel set for version '${key}' (k=${k}, j=${j}, i=${i})")
                    } else {
                        overriddenByLater.addAll(s as Collection)
                    }
                }

                trees << patchP.fileTree(layerJava) {
                    include("**/*.java")
                    exclude(overriddenByLater) // your existing override logic
                    exclude(brokenExcludesForPatch as Collection) // NEW
                }
            }

            // Add older layers, excluding anything overridden by any later layer (including patchVer)
            if (false) {// old logic, kept just in case
                for (int j = 0; j < i; j++) {
                    def v = versions[j]
                    def layerProj = verToProj[v]

                    def overriddenByLater = new LinkedHashSet<String>()
                    for (int k = j + 1; k <= i; k++) {
                        def key = versions[k]
                        def s = relByVer[key]
                        if (s == null) {
                            patchP.logger.lifecycle("[patch-common] WARNING: no rel set for version '${key}' (k=${k}, j=${j}, i=${i})")
                        } else {
                            overriddenByLater.addAll(s as Collection)
                        }
                    }

                    trees << patchP.fileTree(layerProj.file("src/main/java")) {
                        include("**/*.java")
                        exclude(overriddenByLater)
                    }
                }
            }

            // Patch layer last
            trees << patchP.fileTree(patchJava) { include("**/*.java") }

            jc.setSource(patchP.files(trees))
        }


        // ---------- Resources: ordered copy (shared -> all previous -> patch) ----------
        patchP.plugins.withId("java") {
            def resDirs = []
            resDirs << sharedRes

            for (int j = 0; j < i; j++) {
                def v = versions[j]
                resDirs << verToProj[v].file("src/main/resources")
            }

            resDirs << patchRes

            patchP.extensions.configure(SourceSetContainer) { SourceSetContainer ss ->
                ss.named("main") { SourceSet main ->
                    main.resources.setSrcDirs(resDirs)
                }
            }

            patchP.tasks.named("processResources").configure { pr ->
                pr.duplicatesStrategy = DuplicatesStrategy.INCLUDE
            }
        }



    }
}
